Quick Start – Code Examples

// Purchase carbon credits via Harit Setu API
const response = await fetch('https://api.haritsetu.in/v1/credits/purchase', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer hs_live_YOUR_API_KEY',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    quantity_tonnes: 100,
    crop_type: 'paddy',         // or 'coconut', 'sugarcane'
    region: 'Kerala',           // optional filter
    min_ndvi: 0.70,             // quality filter
    retire_immediately: false   // set true to auto-retire
  })
});

const { credit_ids, total_cost, receipt_url } = await response.json();
console.log('Credits purchased:', credit_ids);
import requests

response = requests.post(
    'https://api.haritsetu.in/v1/credits/purchase',
    headers={
        'Authorization': 'Bearer hs_live_YOUR_API_KEY',
        'Content-Type': 'application/json'
    },
    json={
        'quantity_tonnes': 100,
        'crop_type': 'paddy',
        'region': 'Kerala',
        'min_ndvi': 0.70,
        'retire_immediately': False
    }
)

data = response.json()
print('Credits purchased:', data['credit_ids'])
curl -X POST https://api.haritsetu.in/v1/credits/purchase \
  -H "Authorization: Bearer hs_live_YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "quantity_tonnes": 100,
    "crop_type": "paddy",
    "region": "Kerala",
    "min_ndvi": 0.70,
    "retire_immediately": false
  }'
